\begin{frame}{Inference of Structural Causal Models}

The goal is to infer an SCM-model $f_G$ with optimal
\begin{enumerate}[1)]
\item complexity $\loss_\text{C}$,
\item accuracy $\loss_\bD$,
\item robustness $\loss_{\text{ci}}$.
\end{enumerate}

\bigskip
The models:
\begin{enumerate}[1)]
\item the model $\fai$ maps a data set $\bD$ to a model $f_G(\bw,\bx)$,
\item the model $f_G$ maps a sample $\bx\in\bD$ to its forecast $\hat{\bx}$, provided optimal parameters $\hat{\bw}$.
\end{enumerate}
\end{frame}

\begin{frame}{The van der Pol oscillator, the mathematical model}

The forced Van der Pol oscillator:
\[{d^2x \over dt^2}-\mu(1-x^2){dx \over dt}+x-A \sin(\omega t)= 0,\]
has two linear and one non-linear parameters: $\mu, A, \omega$,
where $A$ is the amplitude, of the wave function and $\omega$ is its angular velocity.

The causal graph has four vertices, the variables: 
\[
X_1,X_2,X_3 \mapsto \ddot{x}, \dot{x}, x
\]
and the time $t$.

\begin{frame}{Alternative form of the Van der Pol oscillator model}

An alternative form is a time-independent Hamiltonian by augmenting it to a four-dimensional, $X_1,\dots,X_4$ autonomous dynamical system using an auxiliary second-order nonlinear differential equation:
\[
\ddot{x}-\mu(1-x^2)\dot{x}+x=0, \qquad
\ddot{y}+\mu(1-x^2)\dot{y}+y=0.
\]

The Hamiltonian 
\[
H(x,y,p_x,p_y)=p_xp_y+xy-\mu(1-x^2)yp_y,\]
where
\[
p_x=\dot{y} + \mu(1-x^2)y \quad\text{and}\quad p_y=\dot{x} \]
are the conjugate momenta corresponding to <<$x$>> and <<$y$>>.
{\small}
\end{frame}

\begin{frame}{The basic functions for the Van der Pol model}

The nodes: $X_1(t), X_2(t), X_3$ are the observable variables.%, $t$ is a time observable independent variable.

\bigskip
The latent variables $V_1,\dots,V_6$ are in the set  (the dictionary)
\[
 \{\sin(\cdot), (\cdot)^2, +(w_1,\dots,w_{k+1}), \times(\dots, w_0) \}.
\]

The function <<$+$>> has $k+1$ linear parameters and $k$ arguments. 

The function <<$\times$>> has one parameters and $k^\prime$ arguments. 

\bigskip
The causal model for the graph $G$ 
\[
f_G(\bw): \bx_t \mapsto \hat{\bx}_t \quad \text{with observable variables}\quad (X_2,X_3)(t) \mapsto \hat{X_1}(t)
\]

with parameters $\bw$ and data $\bx$ has the form
\[
X_1(t) = w_1(w_4-w_5 X_3^2) \times X_2(t) - w_2 X_3 + w_3 \sin(w_6 t).
\]

\begin{frame}{The graph generation and rewriting}

A graph $G (V,E)$ with the 
\begin{enumerate}[1)]
\item coloring function $c(V)$, where $c$ is in the dictionary,
\item weights $\bw(V)$,
\item probability of the vertices $p(V)$,
\item probability of the edges $\bZ \in \{0,1\}^{n\times n}$ or $\bGam \in [0,1]^{n\times n}$.
\end{enumerate}

\bigskip
The graph $G$ defined the function $f_G(\bw,\bx)\to\hat{\bx}$. 

\bigskip
There given a graph rewriting rules $G\mapsto G^\prime$ so that each step delivers an admissible graph (admissible means a graph $G$ defines a function $f_G$).

\bigskip
To produce a sequence of the graphs introduce a vector $\bz = \textbf{vec}(\bZ)$.

\begin{frame}{Coherent the SCM inference (problem statement)}

\begin{block}{The given data and the problem}
\begin{enumerate}
\item There given a collection $\{\bD\} \rightarrowtail \{\fG\}$ with set injection. 
\item For any given data set $\bD$ the model $\fai: \bD \mapsto f_G$  such that
\item The model $f_G(\bw): \bx \mapsto \hat{\bx}$ with optimal criteria $(\loss_\text{C},\loss_\bD,\loss_{\text{ci}})$.
\item A context or prior information $p(f_G)$ is provided along with the data $\bD$ as a vector.
\end{enumerate}
\end{block}

\end{frame}
%=============================================
\begin{frame}{Coherent the SCM inference (addendum)}

\begin{block}{Consigns}
\begin{enumerate}
\item Denote the given $\fG$ as $f_0$. 
\item Convert its given graph $G(V,E)(\fc,\fr)$ to the vector $\bz_0$.
\item {The rules $\fr$ rewrite the graph $G$; the colors $\fc$ apply restrictions to the adjacency matrix,~$\bz=\textbf{vec}({\bZ})$, the number of 1s in its row must coincide to the corresponding dictionary function.}
\item The DAG $G$ splits the variables $X_1,\dots,X_n$ to the source and target variables. We denote the corresponding data samples by $\bx\mapsto \hat{\bx}$ without collision. 
\item We treat all types of variables, $X, X(t), X(t,s)$ as random processes, each variable has its own dimensionality. For example, the MHD treat most of variables as a process $X(s,t)$ a vector function $\mathbb{R}^4\to\mathbb{R}^3$.
\end{enumerate}
\end{block}
\end{frame}
%=============================================
\begin{frame}{Coherent the SCM inference (algorithm)}

\begin{block}{The algorithm with steps $t = [T,\dots,0]$}
\begin{enumerate}
\item Sample some new structure $\bz_t$ from the distribution  $q(\bz_t\mid\bz_{t-1})q(\bz_{t-1}\mid\bA_{t-1})q(\fG)$.
\item Convert the structure and its parameters $(\bz,\bw)$ to  the SCM $\fG(\bw,\bx)$.
\item Rewrite the graph $G$ of the SCM according to the rules $\fr$.
\item Note that the initial values of parameters $\bw_t$ of $f_t$ are inherited from $\hat{\bw}_{t-1}$. 
\item Estimate the parameters $\hat{\bw}$ according to the data $\{\bx\}$.
\item Given $\hat{\bw}$ estimate the covariance matrix $\hat{\bA}$.
\item Make do-intervention and estimate the covariance matrix $\hat{\bA}^\prime$.
\item Analyze the couple $\bA,\bA^\prime$ and set the parameters $\bpi$.
\item Repeat until $t=T$. Keep the spectrum of $\bD$ at $t=T$.
\end{enumerate}
\end{block}
\end{frame}
%=============================================
\begin{frame}{(continued)}

\begin{block}{Forecasting}
\begin{enumerate}
\item Given the data set $\bD$ (and its spectrun at $t=T$) and the prior $p(G)$.
\item Make the inference $p_\bth(\bz_{t-1}\mid\bz_{t})$ and compute $\bz_0$.
\item At each step $t$ reconstruct $\bw_t$. No intervention needed.
\item Return $\fG(\hat{\bw},\bx)$. Return optimal $\loss_\bD$, complexity, and robustness.
\end{enumerate}
\end{block}

\vfill
\begin{block}{Notes}
\begin{enumerate}
\item Step 4: new parameters are set to deliver $\text{it}$ to the dictionary functions. Examples. For $+$ the parameters $\bw=[0,0]$ delivers $\text{id}:$ $0x+0=x$. For $+$ the parameters $\bw=[1]$ delivers $\text{id}:$ $1*x = x$. The cross-product $\times$ delivers $\mathbf{I}$ or the identity tensor is written as the Kronecker delta. There is no parameters that delivers $\text{id}$ for functions like $\sin(x), \ln(x)$.
\item Steps 2,3,5,6,7 do not depend on time.
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}{The Metropolis Hastings graph sampling}

Sample the new structure $\bz_t$ from  $q(\bz_t\mid\bz_{t-1})q(\bz_{t-1}\mid\bA_{t-1})q(\fG)$.

\bigskip
To sample graphs~$\bz_0 = \{\bz \mid \bz \sim p(\bz|\boldsymbol{\pi})\}$.
The element $\bz_{t+1}$ of the sample depends on the previous element $\bz_t$ of the sample  only.
Denote by $Q(\bz \mid \bz^\prime)$ an auxiliary distribution $Q(\bz\mid \bz^\prime)$.

Choose an initial element $\bz_0$ and assign $\bz_0 = \{\bz_0\}$.

Then let an element $\bz_t$ be chosen according  to the distribution $Q(\bz^\prime\mid\ba_t)$.
The next element $\bz^\prime$ is generated randomly.
Then the algorithm computes the acceptance ratio $a$:
\[
    a = \min\limits_{\bz^\prime\in\mathbb{R}^n}\left(
    \frac{p(D\mid\ba^{\prime}, B_0)\quad Q(\bz_t\mid\ba^\prime)}
    {p(D\mid\ba_t, B_0)\quad Q(\bz^\prime\mid \bz_t)}, 1\right).
\]

The algorithm accepts the candidate $\bz^\prime$ with the probability $a$,
\[
    \bz_{t+1} = \bz^\prime, \qquad, W_0 := W_0 \cup \bz^\prime.
\]
Otherwise, the algorithm rejects the candidate and puts $\bz_{t+1} = \bz_t$,
\[
\bz_{t+1} =
\begin{cases}
\bz^\prime, \quad\text{with the probability}\quad a, \\
\bz_t,  \quad\text{with the probability} (1-a).
\end{cases}
\]
\end{frame}

%=============================================
\begin{frame}{The Metropolis Hastings graph sampling}
Let the auxiliary distribution $Q(\bz \mid \bz^\prime)$ be normal (it is pretty simplified case; replace to uniform, Dirichlet or Gumbel-Softmax distribution):
\[
    Q(\bz \mid \bz^\prime) = Q(\bz^\prime \mid \bz) = \frac{1}{(2\pi\alpha^{-1})^{n/2}} \exp\left(-\frac{\alpha}{2}(\bz - \bz^\prime)\T (\bz - \bz^\prime)\right).
\]
That is, the function $Q(\bz \mid \bz^\prime)$ is symmetric and
the acceptance ratio $a$ simplifies to
\[
    a = \min\limits_{\bz^\prime\in\mathbb{R}^n}\left(
    \frac{p(D\mid\ba^{\prime}, B_0)}{p(D\mid\ba_t, B_0)}, 1\right).
\]
The initial element $\bz_0$ is chosen randomly from the distribution $p(   \mid \boldsymbol{\pi})$.
\end{frame}

\begin{frame}{The do-inference testing procedure}

\begin{block}{There given}
\begin{enumerate}[1)]
\item the DAG $G$ corresponds to the model $\fG$,
\item its parameters and covariance matrix $(\hat{\bw}, \hat{A})$,
\item the strategy of do-intervention in $\{X\}$ of $n$ steps%
\footnote{For example, set each input $X$ (of $n$ inputs, according to $G$) to the constant $ \text{do}X = \mathbb{E}[X]$.}
\end{enumerate}
\end{block}
 
\begin{block}{Collect the covariance matrices}
\begin{enumerate}
\item Sample a subset of $\bx$ from the dats set $\bD$.
\item Set $\text{do}X$ on the step $1,\dots,n$.
\item Estimate the new covariance matrices $\hat{\bA}_1,\dots,\hat{\bA}_n$ for each step.
\item Estimate the new aggregate covariance matrix $\hat{\bA}$ for each step.
\item Set the parameters for the graph generation $\bpi$. 
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}{Discrete genetic algorithm for model selection}
\begin{enumerate}
    \item There is a set of binary vectors
    $%\[
    \{\mathbf{a}_1,\ldots,\mathbf{a}_P\}, \quad \mathbf{a} \in \{0,1\}^{n}.    $%\]
    \item Select two vectors
    $%\[
    \mathbf{a}_p,\ \mathbf{a}_q,    \quad    p,q \in \{1,\ldots,P\}.    $%\]
    \item Choose a random number
    $%\[
    \nu \in \{1,\ldots,n-1\}.
    $%\]
    \item Split both vectors and exchange their tails:
    \[
    \left[
    a_{p,1},\ldots,a_{p,\nu},
    a_{q,\nu+1},\ldots,a_{q,n}
    \right]
    \rightarrow
    \mathbf{a}'_{p},
    \]
    \[
    \left[
    a_{q,1},\ldots,a_{q,\nu},
    a_{p,\nu+1},\ldots,a_{p,n}
    \right]
    \rightarrow
    \mathbf{a}'_{q}.
    \]
    \item Choose random numbers    $%\[
    \eta_1,\ldots,\eta_Q \in \{1,\ldots,n\}.     $%\]
    \item Invert positions    $%\[
    \eta_1,\ldots,\eta_Q    $ %\]
    of the vectors   $% \[
    \mathbf{a}'_{p},\ \mathbf{a}'_{q}.     $%\]
    \item Repeat Steps 2--6 $\frac{P}{2} $  times.
    \item Evaluate the obtained models.
\end{enumerate}

\bigskip
Repeat the procedure $R$ times; here $P$, $Q$, and $R$ are the parameters of the algorithm, and $n$ is the number of features in the corresponding model.
\end{frame}
